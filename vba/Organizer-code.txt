    For i = Year(Date) - 10 To Year(Date) + 10

    Next i

        If months <> "" Then months = months & "|"

Private Sub ShowStep()
    Me.Controls("stepTitle").Caption = "Reminder " & CStr(StepIndex + 1) & " of " & CStr(StepCount)
    Me.Controls("stepBody").Value = Sequence(StepIndex)
    Panel "sequence"
End Sub

Private Sub SequenceAction(ByVal action As String)
    Dim parts As Variant, i As Long, later As String
    If action = "Sequence" Then
        StepCount = 1
        If CStr(Me.Controls("later_message").Value) <> "" Then
            parts = Split(Replace(CStr(Me.Controls("later_message").Value), vbCrLf, vbLf), vbLf & "---STEP---" & vbLf)
            StepCount = UBound(parts) + 2
        End If
        ReDim Sequence(0 To StepCount - 1)
        Sequence(0) = CStr(Me.Controls("message").Value)
        If StepCount > 1 Then
            For i = 1 To StepCount - 1: Sequence(i) = parts(i - 1): Next i
        End If
        StepIndex = 0
        ShowStep
        Exit Sub
    End If
    Sequence(StepIndex) = CStr(Me.Controls("stepBody").Value)
    Select Case action
        Case "StepPrevious": If StepIndex > 0 Then StepIndex = StepIndex - 1
        Case "StepNext": If StepIndex < StepCount - 1 Then StepIndex = StepIndex + 1
        Case "StepAdd"
            If StepCount >= 100 Then Err.Raise vbObjectError + 2120, , "Maximum 100 messages."
            StepCount = StepCount + 1
            ReDim Preserve Sequence(0 To StepCount - 1)
            Sequence(StepCount - 1) = Sequence(StepIndex)
            StepIndex = StepCount - 1
        Case "StepRemove"
            If StepIndex = 0 Then Err.Raise vbObjectError + 2121, , "The first message is required."
            For i = StepIndex To StepCount - 2: Sequence(i) = Sequence(i + 1): Next i
            StepCount = StepCount - 1
            ReDim Preserve Sequence(0 To StepCount - 1)
            If StepIndex >= StepCount Then StepIndex = StepCount - 1
        Case "StepDone"
            For i = 0 To StepCount - 1
                If Trim$(Sequence(i)) = "" Then Err.Raise vbObjectError + 2122, , "Each reminder message needs text."
            Next i
            Me.Controls("message").Value = Sequence(0)
            For i = 1 To StepCount - 1
                If i > 1 Then later = later & vbLf & "---STEP---" & vbLf
                later = later & Sequence(i)
            Next i
            Me.Controls("later_message").Value = later
            Panel "edit"
            Exit Sub
    End Select
    ShowStep
End Sub

Public Sub ShowDashboard(ByVal view As String)
    Dim row As Object, c As Object, status As String, include As Boolean, index As Long, query As String
    ViewName = view
    If view = "Important" Then
        ShowImportant
        Exit Sub
    End If
    Set Rows = FOCall("list", FOXML("include_removed", "1"))
    Set c = Me.Controls("list")
    query = LCase$(Trim$(CStr(Me.Controls("search").Value)))
    c.Clear
    c.ColumnCount = 8: c.ColumnWidths = "0 pt;150 pt;110 pt;85 pt;85 pt;90 pt;50 pt;135 pt"
    Me.Controls("columns").Caption = "Subject | Recipient | Status | Next (IST) | Last in Sent (IST) | Sent count | Send outcome"
    For Each row In Rows.SelectNodes("row")
        status = FOValue(row, "status")
        include = False
        Select Case view
            Case "Closed": include = (status = "Closed")
            Case "Removed": include = (status = "Removed")
            Case "Exceptions": include = (FOValue(row, "exception") = "1")
            Case "Overdue": include = (FOValue(row, "overdue") = "1")
            Case "Due Today": include = (FOValue(row, "due_today") = "1")
            Case "Review Replies": include = (status = "Review")
            Case "Awaiting Reply": include = (status = "Active" Or status = "Draft ready" Or status = "Paused" Or status = "Pending send")
            Case Else: include = (status <> "Closed" And status <> "Removed")
        End Select
        If query <> "" Then include = include And (InStr(1, LCase$(FOValue(row, "subject") & " " & FOValue(row, "to")), query, vbBinaryCompare) > 0)
        If include Then
            c.AddItem FOValue(row, "id")
            index = c.ListCount - 1
            c.List(index, 1) = FOValue(row, "subject")
            c.List(index, 2) = FOValue(row, "to")
            c.List(index, 3) = status
            c.List(index, 4) = FOValue(row, "next_local")
            c.List(index, 5) = FOValue(row, "last_local")
            c.List(index, 6) = FOValue(row, "submitted_count")
            c.List(index, 7) = FOValue(row, "send_state")
        End If
    Next row
    Me.Controls("status").Caption = view & " | " & CStr(c.ListCount) & " | " & FOValue(Rows, "engine") & " | " & FOValue(Rows, "blocked")
    SetCardCaption "CardOverdue", "Overdue " & FOValue(Rows, "overdue")
    SetCardCaption "CardDue", "Due today " & FOValue(Rows, "due_today")
    SetCardCaption "CardAwaiting", "Awaiting " & FOValue(Rows, "awaiting")
    SetCardCaption "CardReview", "Review " & FOValue(Rows, "review")
    SetCardCaption "CardExceptions", "Exceptions " & FOValue(Rows, "exceptions")
    SetCardCaption "CardRemoved", "Removed " & FOValue(Rows, "removed")
    SetFollowupActions True
    SetActionEnabled "Restore", (view = "Removed")
    SetActionEnabled "Purge", (view = "Removed")
    SetImportantActions False
    Me.Caption = "Follow-up Organizer " & FOValue(Rows, "version")
    Panel "dash"
    If c.ListCount > 0 Then
        c.ListIndex = 0
    Else
        Me.Controls("detailTitle").Caption = "Information"
        Me.Controls("detail").ForeColor = RGB(80, 80, 80)
        Me.Controls("detail").Value = "No follow-ups are available in this view. Times use IST."
    End If
End Sub

Private Function SelectedIDs() As String
    Dim c As Object, i As Long, value As String
    Set c = Me.Controls("list")
    If ViewName = "Conversation" Then
        SelectedIDs = CurrentID
        Exit Function
    End If
    For i = 0 To c.ListCount - 1
        If c.Selected(i) Then
            If value <> "" Then value = value & vbLf
            value = value & CStr(c.List(i, 0))
        End If
    Next i
    SelectedIDs = value
End Function

        Me.Controls("detail").Value = "Setup required" & vbCrLf & _
            "Use Edit / Continue to configure recipients, reminder message and schedule."

    End If

    status = FOValue(result, "status")

    Dim r As Object, key As Variant

        Set FOEditingMail = Application.Session.GetItemFromID(FOValue(r, "anchor_entry"), FOValue(r, "anchor_store"))

Private Sub SearchAndApproveMailboxMail(ByVal recordID As String, ByVal originalMail As Outlook.MailItem, ByVal importantMode As Boolean, ByVal importantCategory As String)
    Dim ns As Outlook.NameSpace, mailbox As Outlook.Store
    Dim inbox As Outlook.Folder, sentFolder As Outlook.Folder, linked As Object, rows As Object, row As Object
    Dim folderCount As Long, messageCount As Long, candidateCount As Long
    Dim linkedCount As Long, skippedCount As Long, errorCount As Long
    Dim stopped As Boolean, stopReason As String, targetSubject As String
    Dim key As String, summary As String, targetConversation As String

    On Error GoTo Failed
    targetSubject = NormalizeRelatedSubject(originalMail.Subject)
    targetConversation = originalMail.ConversationID
    If Len(targetSubject) = 0 And Len(targetConversation) = 0 Then Exit Sub

    Set linked = CreateObject("Scripting.Dictionary")
    linked.CompareMode = vbTextCompare
    If importantMode Then
        Set rows = FOCall("important_links", FOXML("id", recordID))
    Else
        Set rows = FOCall("messages", FOXML("id", recordID))
    End If
    For Each row In rows.SelectNodes("row")
        key = FOValue(row, "store") & "|" & FOValue(row, "entry")
        If Not linked.Exists(key) Then linked.Add key, True
    Next row

    Set ns = Application.Session
    For Each mailbox In ns.Stores
        If stopped Then Exit For
        Set inbox = Nothing
        On Error Resume Next
        Set inbox = mailbox.GetDefaultFolder(olFolderInbox)
        errorCount = errorCount + IIf(Err.Number <> 0, 1, 0)
        Err.Clear
        On Error GoTo Failed
        If Not inbox Is Nothing Then
            ScanRelatedFolder inbox, recordID, targetSubject, targetConversation, linked, importantMode, importantCategory, _
                              folderCount, messageCount, candidateCount, _
                              linkedCount, skippedCount, errorCount, _
                              stopped, stopReason
        End If
        If stopped Then Exit For
        Set sentFolder = Nothing
        On Error Resume Next
        Set sentFolder = mailbox.GetDefaultFolder(olFolderSentMail)
        errorCount = errorCount + IIf(Err.Number <> 0, 1, 0)
        Err.Clear
        On Error GoTo Failed
        If Not sentFolder Is Nothing Then
            ScanRelatedFolder sentFolder, recordID, targetSubject, targetConversation, linked, importantMode, importantCategory, _
                              folderCount, messageCount, candidateCount, _
                              linkedCount, skippedCount, errorCount, _
                              stopped, stopReason
        End If
    Next mailbox

    summary = "Related-subject search completed." & vbCrLf & vbCrLf & _
              "Candidates reviewed: " & CStr(candidateCount) & vbCrLf & _
              "Emails linked: " & CStr(linkedCount) & vbCrLf & _
              "Emails skipped: " & CStr(skippedCount) & vbCrLf & _
              "Folders searched: " & CStr(folderCount) & vbCrLf & _
              "Messages inspected: " & CStr(messageCount)
    If errorCount > 0 Then summary = summary & vbCrLf & "Items/folders not readable: " & CStr(errorCount)
    If stopped Then summary = summary & vbCrLf & vbCrLf & "Search stopped: " & stopReason
    MsgBox summary, vbInformation, "Related mailbox mail"
    Exit Sub

Failed:
    If importantMode Then
        MsgBox "The bookmark was saved, but the related-mail search stopped." & vbCrLf & Err.Description, vbExclamation, "Related mailbox mail"
    Else
        MsgBox "The follow-up was saved, but the related-mail search stopped." & vbCrLf & Err.Description, vbExclamation, "Related mailbox mail"
    End If
End Sub

Private Sub ScanRelatedFolder( _
    ByVal sourceFolder As Outlook.Folder, _
    ByVal followupID As String, _
    ByVal targetSubject As String, _
    ByVal targetConversation As String, _
    ByVal linked As Object, _
    ByVal importantMode As Boolean, _
    ByVal importantCategory As String, _
    ByRef folderCount As Long, _
    ByRef messageCount As Long, _
    ByRef candidateCount As Long, _
    ByRef linkedCount As Long, _
    ByRef skippedCount As Long, _
    ByRef errorCount As Long, _
    ByRef stopped As Boolean, _
    ByRef stopReason As String)

    Dim items As Outlook.Items, item As Object, mail As Outlook.MailItem
    Dim child As Outlook.Folder, key As String, prompt As String
    Dim answer As VbMsgBoxResult, result As Object, i As Long, linkedConfirmed As Boolean

    If stopped Then Exit Sub
    folderCount = folderCount + 1
    If folderCount > RELATED_MAX_FOLDERS Then
        stopped = True: stopReason = "folder safety limit reached"
        Exit Sub
    End If

    On Error GoTo FolderFailed
    Set items = sourceFolder.Items
    items.Sort "[ReceivedTime]", True

    For i = 1 To items.Count
        If stopped Then Exit For
        On Error GoTo ItemFailed
        Set item = items.Item(i)
        If TypeOf item Is Outlook.MailItem Then
            messageCount = messageCount + 1
            If messageCount > RELATED_MAX_MESSAGES Then
                stopped = True: stopReason = "message safety limit reached"
                Exit For
            End If

            Set mail = item
            key = mail.Parent.StoreID & "|" & mail.EntryID
            If Not linked.Exists(key) And FOProp(mail, "FOGenerated") = "" Then
                If Not importantMode Then
                    If FOProp(mail, "FOID") <> "" And FOProp(mail, "FOID") <> followupID Then GoTo NextItem
                End If
                If (targetConversation <> "" And mail.ConversationID = targetConversation) Or _
                   (targetSubject <> "" And NormalizeRelatedSubject(mail.Subject) = targetSubject) Then
                    candidateCount = candidateCount + 1
                    If candidateCount > RELATED_MAX_CANDIDATES Then
                        stopped = True: stopReason = "candidate approval limit reached"
                        Exit For
                    End If

                    If importantMode Then
                        prompt = "Link this email to the new Important Mail bookmark?"
                    Else
                        prompt = "Link this email to the new follow-up?"
                    End If
                    prompt = prompt & vbCrLf & vbCrLf & _
                             "Subject: " & mail.Subject & vbCrLf & _
                             "Received: " & Format$(mail.ReceivedTime, "dd mmm yyyy hh:nn") & " IST" & vbCrLf & _
                             "Mailbox: " & mail.Parent.Store.DisplayName & vbCrLf & _
                             "Folder: " & sourceFolder.FolderPath & vbCrLf & vbCrLf & _
                             "Yes = Link | No = Skip | Cancel = Stop search"
                    answer = MsgBox(prompt, vbYesNoCancel + vbQuestion, "Related Inbox mail")
                    linkedConfirmed = False

                    If answer = vbCancel Then
                        stopped = True: stopReason = "cancelled by user"
                        Exit For
                    ElseIf answer = vbYes Then
                        If importantMode Then
                            Set result = FOCall("important_attach", FOXML("id", followupID) & FOSnapshot(mail, "a_"))
                        Else
                            Set result = FOCall("attach_related", FOXML("id", followupID) & FOSnapshot(mail, "a_"))
                        End If
                        linkedConfirmed = (LCase$(FOValue(result, "linked")) = "true")
                        If linkedConfirmed Then
                            linkedCount = linkedCount + 1
                            If Not linked.Exists(key) Then linked.Add key, True
                            If importantMode Then
                                MarkImportantMail mail, importantCategory
                            Else
                                FOPut mail, "FOID", followupID
                                mail.Save
                            End If
                        Else
                            skippedCount = skippedCount + 1
                        End If
                    Else
                        skippedCount = skippedCount + 1
                    End If
                End If
            End If
        End If

NextItem:
        Set mail = Nothing: Set item = Nothing
        On Error GoTo FolderFailed
    Next i

    If Not stopped Then
        For Each child In sourceFolder.Folders
            ScanRelatedFolder child, followupID, targetSubject, targetConversation, linked, importantMode, importantCategory, _
                              folderCount, messageCount, candidateCount, _
                              linkedCount, skippedCount, errorCount, _
                              stopped, stopReason
            If stopped Then Exit For
        Next child
    End If
    Exit Sub

ItemFailed:
    errorCount = errorCount + 1
    Err.Clear
    Resume NextItem

FolderFailed:
    errorCount = errorCount + 1
    Err.Clear
End Sub

Private Function NormalizeRelatedSubject(ByVal value As String) As String
    Dim text As String, changed As Boolean, ch As String, clean As String
    Dim i As Long, prefix As Variant

    text = LCase$(Trim$(value))
    Do
        changed = False
        For Each prefix In Array("re:", "fw:", "fwd:")
            If Left$(text, Len(CStr(prefix))) = CStr(prefix) Then
                text = Trim$(Mid$(text, Len(CStr(prefix)) + 1))
                changed = True
                Exit For
            End If
        Next prefix
    Loop While changed

    For i = 1 To Len(text)
        ch = Mid$(text, i, 1)
        If ch Like "[a-z0-9]" Then
            clean = clean & ch
        ElseIf Right$(clean, 1) <> " " Then
            clean = clean & " "
        End If
    Next i
    NormalizeRelatedSubject = Trim$(clean)
End Function

Private Function ImportantFields(Optional ByVal existing As Object = Nothing) As String
    Dim categories As Object, category As String, priority As String, tags As String, note As String, reviewDate As String
    Set categories = FOCall("important_categories")
    If existing Is Nothing Then
        category = "Quality": priority = "Normal"
    Else
        category = FOValue(existing, "category"): priority = FOValue(existing, "priority")
        tags = FOValue(existing, "tags"): note = FOValue(existing, "note"): reviewDate = FOValue(existing, "review_date")
    End If
    category = Trim$(InputBox("Choose an existing category or type a new one:" & vbCrLf & _
        Replace(FOValue(categories, "categories"), "|", ", "), "Important Mail category", category))
    If category = "" Then Err.Raise vbObjectError + 2160, , "A category is required."
    priority = Trim$(InputBox("Priority: Normal, High or Critical", "Important Mail priority", priority))
    If priority = "" Then priority = "Normal"
    tags = InputBox("Optional search tags, separated by semicolons:", "Important Mail tags", tags)
    note = InputBox("Optional private note (stored only in Organizer):", "Important Mail note", note)
    reviewDate = InputBox("Optional review date as YYYY-MM-DD:", "Important Mail review", reviewDate)
    ImportantFields = FOXML("category", category) & FOXML("priority", priority) & FOXML("tags", tags) & _
        FOXML("note", note) & FOXML("review_date", Trim$(reviewDate))
End Function

Private Sub MarkImportantMail(ByVal mail As Outlook.MailItem, ByVal category As String)
    Dim value As String
    value = mail.Categories
    If InStr(1, value, "FO Important", vbTextCompare) = 0 Then
        If value <> "" Then value = value & ", "
        value = value & "FO Important"
    End If
    If InStr(1, value, "FO " & category, vbTextCompare) = 0 Then value = value & ", FO " & category
    mail.Categories = value
    mail.Save
End Sub

            c.ColumnCount = 5: c.ColumnWidths = "0 pt;245 pt;190 pt;110 pt;170 pt"
            Me.Controls("columns").Caption = "Subject | Received (IST) | Action | Store"
            For Each row In Rows.SelectNodes("row")
                c.AddItem FOValue(row, "entry")
                i = c.ListCount - 1
                c.List(i, 1) = FOValue(row, "subject")
                c.List(i, 2) = FOValue(row, "received")
                c.List(i, 3) = "Open mail"
                c.List(i, 4) = FOValue(row, "store")
            Next row
            ViewName = "Conversation"
            Me.Controls("status").Caption = "Conversation: select a message and click Open mail / draft"
            Me.Controls("detailTitle").Caption = "Conversation"
            Me.Controls("detail").ForeColor = RGB(0, 0, 110)
            Me.Controls("detail").Value = "Select a linked email above and click Open mail / draft. Conversation dates are shown newest first in IST."
        Case "Open"
            Set c = Me.Controls("list")
            If c.ListIndex < 0 Then Err.Raise vbObjectError + 2112, , "Select a message or follow-up."
            If ViewName = "Important" Then
                On Error Resume Next
                Set mail = ImportantMailFromRow()
                If mail Is Nothing Then
                    Err.Clear: On Error GoTo Failed
                    Set result = FOCall("important_missing", FOXML("ids", SelectedID()))
                    Err.Raise vbObjectError + 2165, , "The Outlook email could not be found. The bookmark is marked Missing; use Relink selected."
                End If
                On Error GoTo Failed
            ElseIf ViewName = "Conversation" Then
                Set mail = Application.Session.GetItemFromID(c.List(c.ListIndex, 0), c.List(c.ListIndex, 4))
            Else
                id = SelectedID()
                Set result = FOCall("messages", FOXML("id", id))
                ' The messages endpoint is newest-first; open the newest linked mail.
                Set row = result.SelectSingleNode("row[1]")
                If row Is Nothing Then Err.Raise vbObjectError + 2113, , "Original is not yet available in Sent Items."
                Set mail = Application.Session.GetItemFromID(FOValue(row, "entry"), FOValue(row, "store"))
            End If
            mail.Display
        Case "Pause", "Close", "Later", "Remove"
            text = SelectedIDs()
            If text = "" Then Err.Raise vbObjectError + 2123, , "Select at least one follow-up."
            If action = "Remove" Then
                If MsgBox("Remove the selected follow-up(s) from Organizer?" & vbCrLf & vbCrLf & _
                          "No further reminders will be created. Outlook emails will not be deleted." & vbCrLf & _
                          "Any existing Draft or Outbox item must still be checked manually.", _
                          vbYesNo + vbExclamation, "Remove follow-up") <> vbYes Then Exit Sub
            End If
            Set result = FOCall("bulk", FOXML("ids", text) & FOXML("method", action))
            ShowDashboard "Today"
        Case "Restore"
            text = SelectedIDs()
            If text = "" Then Err.Raise vbObjectError + 2145, , "Select at least one removed follow-up."
            If MsgBox("Restore the selected follow-up(s) as Paused?" & vbCrLf & _
                      "Review their schedules and existing Draft/Outbox items before continuing.", _
                      vbYesNo + vbQuestion, "Restore follow-up") = vbYes Then
                Set result = FOCall("restore", FOXML("ids", text))
                ShowDashboard "Removed"
            End If
        Case "Purge"
            text = SelectedIDs()
            If text = "" Then Err.Raise vbObjectError + 2146, , "Select at least one removed follow-up."
            If InputBox("This permanently deletes the selected Organizer records, linked-message metadata and history." & _
                        vbCrLf & "Outlook emails are not deleted." & vbCrLf & vbCrLf & _
                        "Type PURGE to continue:", "Permanent purge") = "PURGE" Then
                Set result = FOCall("purge", FOXML("ids", text) & FOXML("confirm", "PURGE"))
                ShowDashboard "Removed"
            End If
        Case "Resolve"
            id = SelectedID()
            If MsgBox("Check Sent Items, Outbox and Drafts first. Remove any unsent organizer draft you no longer want. Mark this occurrence reviewed without retrying it?", vbYesNo + vbExclamation) = vbYes Then
                Set result = FOCall("resolve", FOXML("id", id))
                ShowDashboard "Today"
            End If
        Case "Link"
            id = SelectedID()
            Set mail = FOSelected()
            If mail Is Nothing Then Err.Raise vbObjectError + 2114, , "Select one Outlook email to link."
            Set result = FOCall("link", FOXML("id", id) & FOSnapshot(mail, "a_"))
            FOPut mail, "FOID", id: mail.Save
            MsgBox "Email linked. Review schedule before resuming.", vbInformation
        Case "SaveSettings"
            For Each key In Split("weekdays|start|end|morning|evening|holidays", "|")
                fields = fields & FOXML(CStr(key), CStr(Me.Controls("s_" & CStr(key)).Value))
            Next key
            If Me.Controls("s_auto").Value Then
                If MsgBox("Enable Auto-mode reminder sending after completing the Windows pilot checklist?", vbYesNo + vbQuestion) <> vbYes Then Exit Sub
                fields = fields & FOXML("auto_enabled", "1")
            End If
            If Me.Controls("s_catchup").Value Then
                fields = fields & FOXML("catchup_enabled", "1")
            Else
                fields = fields & FOXML("catchup_enabled", "0")
            End If
            fields = fields & FOXML("catchup_hours", CStr(Me.Controls("s_catchup_hours").Value))
            Set result = FOCall("save_settings", fields)
            MsgBox "Preferences saved.", vbInformation
        Case "Resume"
            If MsgBox("Review safety reason and pending send outcomes first. Resume eligible follow-ups?", vbYesNo + vbQuestion) = vbYes Then
                Set result = FOCall("resume")
                ShowDashboard "Today"
            End If
        Case "Folder": MonitorFollowupFolder
        Case "Export": ExportFollowupsToExcel
        Case "Import": ImportFollowupsFromExcel
    End Select
    Exit Sub
Failed:
    MsgBox Err.Description, vbExclamation, "Follow-up Organizer"
End Sub

Private Sub BeginUnlink()
    Dim c As Object, r As Object, row As Object, i As Long
    If ViewName <> "Conversation" Then Err.Raise vbObjectError + 2140, , "Open Conversation, select the email, then click Unlink email."
    Set c = Me.Controls("list")
    If c.ListIndex < 0 Then Err.Raise vbObjectError + 2141, , "Select the email to unlink."
    UnlinkEntry = c.List(c.ListIndex, 0): UnlinkStore = c.List(c.ListIndex, 4)
    Me.Controls("unlinkTitle").Caption = "Unlink: " & c.List(c.ListIndex, 1)
    Set r = FOCall("get", FOXML("id", CurrentID))
    CurrentVersion = FOValue(r, "version")
    Set r = FOCall("messages", FOXML("id", CurrentID))
    Set c = Me.Controls("replacement"): c.Clear
    c.AddItem "": c.List(0, 1) = "No replacement (only when this is not the reply source)": c.List(0, 2) = ""
    For Each row In r.SelectNodes("row")
        If Not (FOValue(row, "entry") = UnlinkEntry And FOValue(row, "store") = UnlinkStore) Then
            If FOValue(row, "generated") = "" And FOValue(row, "occ") = "" Then
                c.AddItem FOValue(row, "entry"): i = c.ListCount - 1
                c.List(i, 1) = FOValue(row, "subject") & " | " & FOValue(row, "received")
                c.List(i, 2) = FOValue(row, "store")
            End If
        End If
    Next row
    c.ListIndex = 0
    Panel "unlink"
End Sub

Private Sub ChooseContacts()
    Dim dialog As Outlook.SelectNamesDialog, recipient As Outlook.Recipient
    Dim key As Variant, value As Variant, address As String, toText As String, ccText As String
    Dim seen As Object, r As Object, message As String
    On Error GoTo Failed
    Set r = FOCall("ui_busy", FOXML("busy", "1"))
    Set dialog = Application.Session.GetSelectNamesDialog
    dialog.NumberOfRecipientSelectors = olShowToCc
    dialog.AllowMultipleSelection = True
    dialog.ForceResolution = True
    dialog.Caption = "Follow-up recipients - Contacts / Address Book"
    For Each key In Array("to", "cc")
        For Each value In Split(CStr(Me.Controls(CStr(key)).Value), ";")
            If Trim$(CStr(value)) <> "" Then
                Set recipient = dialog.Recipients.Add(Trim$(CStr(value)))
                If key = "to" Then recipient.Type = olTo Else recipient.Type = olCC
            End If
        Next value
    Next key
    If dialog.Display Then
        Set seen = CreateObject("Scripting.Dictionary"): seen.CompareMode = vbTextCompare
        For Each recipient In dialog.Recipients
            address = ContactSMTP(recipient)
            If Not seen.Exists(address) Then
                seen.Add address, True
                If recipient.Type = olTo Then
                    If toText <> "" Then toText = toText & ";"
                    toText = toText & address
                ElseIf recipient.Type = olCC Then
                    If ccText <> "" Then ccText = ccText & ";"
                    ccText = ccText & address
                End If
            End If
        Next recipient
        Me.Controls("to").Value = toText: Me.Controls("cc").Value = ccText
    End If
Done:
    ' Preserve the outer before-send dialog lease when adding from compose.
    On Error Resume Next
    If Not InCompose Then Set r = FOCall("ui_busy", FOXML("busy", "0"))
    On Error GoTo 0
    If message <> "" Then MsgBox message, vbExclamation, "Outlook Contacts"
    Exit Sub
Failed:
    message = Err.Description
    Resume Done
End Sub

Private Function ContactSMTP(ByVal recipient As Outlook.Recipient) As String
    Dim entry As Outlook.AddressEntry, ex As Outlook.ExchangeUser, dl As Outlook.ExchangeDistributionList
    If Not recipient.Resolved Then Err.Raise vbObjectError + 2142, , "Resolve each recipient in the address book."
    Set entry = recipient.AddressEntry
    If entry.Type = "EX" Then
        Set ex = entry.GetExchangeUser
        If Not ex Is Nothing Then ContactSMTP = ex.PrimarySmtpAddress
        If ContactSMTP = "" Then
            Set dl = entry.GetExchangeDistributionList
            If Not dl Is Nothing Then ContactSMTP = dl.PrimarySmtpAddress
        End If
    Else
        ContactSMTP = entry.Address
    End If
    If InStr(1, ContactSMTP, "@") < 2 Then Err.Raise vbObjectError + 2143, , "No SMTP email address for " & recipient.Name & ". Choose an email-enabled contact."
End Function
